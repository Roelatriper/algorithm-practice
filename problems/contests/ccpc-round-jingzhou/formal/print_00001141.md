#include <iostream>
#include <vector>
#include <cmath>

using namespace std;
#define ll long long

int main(){
    freopen("in.txt","r",stdin);
    ll int t;
    cin >> t;
    for(ll int i=0;i<t;i++){
        ll int n;
        cin >> n;
        ll int pos_p[2],pos_q[2];
        cin >> pos_p[0] >> pos_p[1] >> pos_q[0] >> pos_q[1];
        vector<vector<ll int> > a(n,vector<ll int>(2));
        for(ll int i=0;i<n;i++){
            for(ll int j=0;j<2;j++)cin >> a[i][j];
        }
        ll int cnt = 0;
        for(ll int k_1=0;k_1<n-2;k_1++){
            for(ll int k_2=k_1+1;k_2<n-1;k_2++){
                for(ll int k_3=k_2+1;k_3<n;k_3++){
                    ll int tranagle_abp=
                    abs((a[k_1][0]-pos_p[0])*(a[k_2][1]-pos_p[1])-(a[k_2][0]-pos_p[0])*(a[k_1][1]-pos_p[1]));
                    ll int tranagle_abc=
                    abs((a[k_1][0]-a[k_3][0])*(a[k_2][1]-a[k_3][1])-(a[k_2][0]-a[k_3][0])*(a[k_1][1]-a[k_3][1]));
                    ll int tranagle_bcp=
                    abs((a[k_2][0]-pos_p[0])*(a[k_3][1]-pos_p[1])-(a[k_3][0]-pos_p[0])*(a[k_2][1]-pos_p[1]));
                    ll int tranagle_abq=
                    abs((a[k_1][0]-pos_q[0])*(a[k_2][1]-pos_q[1])-(a[k_2][0]-pos_q[0])*(a[k_1][1]-pos_q[1]));
                    ll int tranagle_bcq=
                    abs((a[k_2][0]-pos_q[0])*(a[k_3][1]-pos_q[1])-(a[k_3][0]-pos_q[0])*(a[k_2][1]-pos_q[1]));

                    cout << k_1 << endl << k_2 << endl << k_3 << endl;
                    cout << "abp:" << tranagle_abp << endl;
                    cout << "abc:" << tranagle_abc << endl;
                    cout << "bcp:" << tranagle_bcp << endl;
                    cout << "abq:" << tranagle_abq << endl;
                    cout << "bcq:" << tranagle_bcq << endl;
                    bool istrue_1=tranagle_abp + tranagle_bcp < tranagle_abc ? 1 : 0;
                    bool istrue_2=tranagle_abq + tranagle_bcq < tranagle_abc ? 1 : 0;
                    if((istrue_1 ^ istrue_2) == 1)cnt++;
                }
            }
        }
        cout << cnt << endl;
    }
    return 0;
}
